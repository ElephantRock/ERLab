# backend/services/arxiv_service.py

class ArxivService:
    def fetch_daily_papers(self, categories=['cs.AI', 'cs.CL', 'cs.LG', 'cs.CV']):
        """Fetch new papers from arXiv API"""
        base_url = "http://export.arxiv.org/api/query"
        yesterday = (datetime.now() - timedelta(days=1)).strftime("%Y%m%d")
        
        papers = []
        for category in categories:
            query = f"cat:{category} AND submittedDate:[{yesterday}0000 TO {yesterday}2359]"
            response = requests.get(base_url, params={
                'search_query': query,
                'start': 0,
                'max_results': 2000,
                'sortBy': 'submittedDate',
                'sortOrder': 'descending'
            })
            feed = feedparser.parse(response.text)
            papers.extend(self._parse_entries(feed.entries))
        
        return papers
    
    def download_pdf(self, paper_id, save_path):
        """Download PDF from arXiv"""
        url = f"https://arxiv.org/pdf/{paper_id}.pdf"
        response = requests.get(url, stream=True)
        with open(save_path, 'wb') as f:
            for chunk in response.iter_content(chunk_size=8192):
                f.write(chunk)
        return save_path
```

# backend/services/pdf_service.py

class PDFService:
    def extract_text_and_structure(self, pdf_path):
        """Convert PDF to structured markdown with metadata"""
        
        # Use marker for high-quality extraction
        import marker
        
        full_text = marker.convert(pdf_path)
        
        # Extract sections
        sections = self._parse_sections(full_text)
        
        # Extract figures with captions
        figures = self._extract_figures(pdf_path)
        
        # Extract tables
        tables = self._extract_tables(full_text)
        
        # Extract references
        references = self._extract_references(full_text)
        
        return {
            "full_text": full_text,
            "sections": sections,
            "figures": figures,
            "tables": tables,
            "references": references,
            "metadata": self._extract_metadata(full_text)
        }
    
    def chunk_for_embedding(self, structured_text):
        """Create semantic chunks for vector embedding"""
        chunks = []
        
        # Chunk by section with overlap
        for section in structured_text['sections']:
            section_text = section['content']
            
            # Split long sections into smaller chunks
            if len(section_text) > 2000:
                sub_chunks = self._semantic_split(section_text, 
                                                   chunk_size=1000, 
                                                   overlap=200)
                for i, chunk in enumerate(sub_chunks):
                    chunks.append({
                        'text': chunk,
                        'metadata': {
                            'section': section['title'],
                            'chunk_index': i,
                            'paper_id': structured_text['metadata']['paper_id']
                        }
                    })
            else:
                chunks.append({
                    'text': section_text,
                    'metadata': {
                        'section': section['title'],
                        'paper_id': structured_text['metadata']['paper_id']
                    }
                })
        
        return chunks
```

        scored_papers = []

                score += 30

            similarity = self._calculate_interest_similarity(paper.abstract)
            if similarity > self.rules['semantic_threshold']:
                score += similarity * 20
                reasons.append(f"Interest similarity: {similarity:.2f}")
            
            # 3. Author/venue prestige (optional)
            if paper.has_influential_authors():
                score += 10
                reasons.append("Influential author")
            
            # 4. Citation velocity (if available)
            velocity = self._get_citation_velocity(paper.id)
            if velocity > threshold:
                score += velocity * 5
                reasons.append(f"Citation velocity: {velocity}")
            
            paper.curation_score = score
            paper.curation_reasons = reasons
            scored_papers.append(paper)
        
        # Sort by score and limit
        scored_papers.sort(key=lambda p: p.curation_score, reverse=True)
        return scored_papers[:self.rules['max_papers_per_day']]

# backend/agents/verifiers/wiki_verifier.py

class WikiVerifier:
    def verify_wiki_accuracy(self, wiki_entry, full_text):
        """Verify that wiki claims are supported by the paper text"""
        
        # Extract factual claims from wiki
        claims = self._extract_factual_claims(wiki_entry)
        
        # Verify each claim against the source text
        verification_results = []
        for claim in claims:
            is_supported = self._verify_claim_against_text(claim, full_text)
            verification_results.append({
                'claim': claim,
                'supported': is_supported,
                'confidence': 'high' if is_supported else 'low'
            })
        
        # Flag if too many unsupported claims
        unsupported_ratio = sum(1 for r in verification_results 
                               if not r['supported']) / len(verification_results)
        
        if unsupported_ratio > 0.2:
            wiki_entry['needs_human_review'] = True
            wiki_entry['verification_notes'] = f"{unsupported_ratio:.0%} claims unsupported"
        
        return wiki_entry
```

# backend/agents/detectors/method_problem_detector.py

class MethodProblemGapDetector(BaseGapDetector):
    def detect(self):
        # Get all methods from claims
        methods = self.db.query("""
            SELECT DISTINCT 
                c.content->>'name' as method_name,
                c.content->>'category' as category,
                c.content->'constraints' as constraints,
                c.content as full_claim,
                p.paper_id,
                w.wiki_content
            FROM claims c
            JOIN wiki_entries w ON c.paper_id = w.paper_id
            JOIN papers p ON c.paper_id = p.id
            WHERE c.claim_type = 'method'
        """).fetchall()
        
        # Get all problems/datasets
        problems = self.db.query("""
            SELECT DISTINCT
                jsonb_array_elements_text(c.content->'datasets') as dataset,
                c.content as full_claim,
                p.paper_id,
                w.wiki_content
            FROM claims c
            JOIN wiki_entries w ON c.paper_id = w.paper_id
            JOIN papers p ON c.paper_id = p.id
            WHERE c.claim_type = 'method'
        """).fetchall()
        
        # Build existing pairs matrix
        existing_pairs = set()
        for claim in self.db.query("SELECT content FROM claims WHERE claim_type='result'").fetchall():
            method = claim['content'].get('method_name')
            dataset = claim['content'].get('dataset')
            if method and dataset:
                existing_pairs.add((method, dataset))
        
        # Find gaps
        gaps = []
        for method in methods:
            method_datasets = self._infer_applicable_datasets(method)
            
            for problem in problems:
                if (method['method_name'], problem['dataset']) not in existing_pairs:
                    # Check if method is applicable to this problem type
                    applicability = self._assess_applicability(method, problem)
                    
                    if applicability['score'] > 0.5:  # Threshold
                        gap = {
                            "gap_type": "method_problem",
                            "method": {
                                "name": method['method_name'],
                                "category": method['category'],
                                "source_paper": method['paper_id'],
                                "constraints": method['constraints']
                            },
                            "problem": {
                                "dataset": problem['dataset'],
                                "source_paper": problem['paper_id']
                            },
                            "applicability_score": applicability['score'],
                            "applicability_rationale": applicability['rationale'],
                            "potential_adaptations_needed": applicability['adaptations'],
                            "estimated_difficulty": applicability['difficulty']
                        }
                        
                        scores = self.score_gap(gap)
                        gap.update(scores)

# backend/agents/detectors/scale_gap_detector.py

class ScaleGapDetector(BaseGapDetector):
    def detect(self):
        # Find methods with scaling data
        methods_with_scaling = self.db.query("""
            SELECT 
                content->>'method_name' as method_name,
                jsonb_agg(jsonb_build_object(
                    'scale', content->>'scale_variable',
                    'value', content->>'value',
                    'dataset', content->>'dataset'
                )) as scaling_data,
                MAX((content->>'value')::float) as max_scale_tested
            FROM claims
            WHERE claim_type = 'result' 
            AND content->>'scale_variable' IS NOT NULL
            GROUP BY content->>'method_name'
            HAVING COUNT(*) >= 3
        """).fetchall()
        
        # Find largest datasets available
        largest_datasets = self.db.query("""
            SELECT 
                content->>'dataset' as dataset,
                MAX((content->>'sample_size')::int) as max_samples
            FROM claims
            WHERE claim_type = 'result'
            AND content->>'sample_size' IS NOT NULL
            GROUP BY content->>'dataset'
            ORDER BY max_samples DESC
        """).fetchall()
        
        gaps = []
        for method in methods_with_scaling:
            for dataset_info in largest_datasets:
                # Check if this method has been tested at this scale
                max_tested = max(
                    (d['value'] for d in method['scaling_data'] 
                     if d['dataset'] == dataset_info['dataset']),
                    default=0
                )
                
                scale_ratio = dataset_info['max_samples'] / max_tested if max_tested > 0 else float('inf')
                
                if scale_ratio > 10:  # Method tested on <10% of available scale
                    # Fit scaling law
                    scaling_points = [
                        (d['value'], d.get('performance', 0)) 
                        for d in method['scaling_data']
                    ]
                    projected_performance = self._extrapolate_performance(
                        scaling_points, dataset_info['max_samples']
                    )
                    
                    gap = {
                        "gap_type": "scale",
                        "method": method['method_name'],
                        "dataset": dataset_info['dataset'],
                        "max_scale_tested": max_tested,
                        "max_scale_available": dataset_info['max_samples'],
                        "scale_ratio": scale_ratio,
                        "scaling_points": scaling_points,
                        "projected_performance": projected_performance,
                        "extrapolation_uncertainty": "high" if scale_ratio > 100 else "medium"
                    }
                    
                    scores = self.score_gap(gap)
                    gap.update(scores)
                    gap['title'] = self.generate_gap_title(gap)
                    
                    gaps.append(gap)
        
        return sorted(gaps, key=lambda g: g.get('total_score', 0), reverse=True)
    
    def _extrapolate_performance(self, points, target_scale):
        """Fit power law and extrapolate"""
        import numpy as np
        from scipy.optimize import curve_fit
        
        if len(points) < 3:
            return None
        
        x = np.array([p[0] for p in points])
        y = np.array([p[1] for p in points])
        
        def power_law(x, a, b, c):
            return a * x ** (-b) + c
        
        try:
            params, _ = curve_fit(power_law, x, y, maxfev=5000)
            projected = power_law(target_scale, *params)
            return {
                'value': float(projected),
                'params': {'a': float(params[0]), 'b': float(params[1]), 'c': float(params[2])},
                'r_squared': self._r_squared(y, power_law(x, *params))
            }
        except:
            return None

class ExperimentRunner:
    def __init__(self, sandbox_path="/sandbox"):
        self.sandbox_path = sandbox_path
        self.docker_client = docker.from_env()
    
    def run_experiment(self, experiment_config):
        """Run an experiment in an isolated Docker container"""
        
        # Prepare experiment directory
        exp_dir = Path(self.sandbox_path) / experiment_config['experiment_id']
        exp_dir.mkdir(parents=True, exist_ok=True)
        
        # Write code to file
        code_file = exp_dir / "experiment.py"
        code_file.write_text(experiment_config['code'])
        
        # Write requirements if provided
        if experiment_config.get('requirements'):
            req_file = exp_dir / "requirements.txt"
            req_file.write_text(experiment_config['requirements'])
        
        # Write any data files
        if experiment_config.get('data_files'):
            for data_file in experiment_config['data_files']:
                dest = exp_dir / data_file['name']
                dest.write_text(data_file['content'])
        
        # Build Docker command
        container = self.docker_client.containers.run(
            image="experiment-sandbox:latest",
            command=f"python /sandbox/experiment.py",
            volumes={
                str(exp_dir): {'bind': '/sandbox', 'mode': 'rw'}
            },
            environment=experiment_config.get('env_vars', {}),
            detach=True,
            mem_limit=experiment_config.get('memory_limit', '8g'),
            nano_cpus=int(experiment_config.get('cpu_limit', 4)) * 1e9,
            device_requests=[
                docker.types.DeviceRequest(
                    device_ids=experiment_config.get('gpu_ids', ['0']),
                    capabilities=[['gpu']]
                )
            ] if experiment_config.get('use_gpu') else None
        )
        
        # Monitor and collect results
        result = container.wait()
        logs = container.logs().decode('utf-8')
        container.remove()
        
        # Parse outputs
        outputs = self._collect_outputs(exp_dir)
        
        return {
            'exit_code': result['StatusCode'],
            'logs': logs,
            'outputs': outputs,
            'artifacts': self._collect_artifacts(exp_dir)
        }
    
    def _collect_outputs(self, exp_dir):
        """Collect structured outputs from experiment"""
        outputs = {}

        # Look for figures

class ResearchPipelineOrchestrator:
    """Coordinates the entire PDF→Paper pipeline"""
    
    def __init__(self):
        self.db = Database()
        self.arxiv = ArxivService()
        self.pdf = PDFService()
        self.curation = CurationService()
        self.llm = LLMService()
        self.vector = VectorService()
        
        self.wiki_extractor = WikiExtractor(self.llm)
        self.claim_extractor = ClaimExtractor(self.llm)
        self.artifact_generator = ArtifactGenerator(self.db)
        
        self.gap_detectors = [
            MethodProblemGapDetector(self.db, self.llm),
            ContradictionDetector(self.db, self.llm),
            ScaleGapDetector(self.db, self.llm),
        ]
        
        self.study_generator = StudyGenerator(self.llm, self.db)
        self.experiment_manager = ExperimentManager()
        self.paper_generator = PaperGenerator(self.llm)
        self.paper_refiner = PaperRefinementAgent(self.llm)
    
    def daily_pipeline(self):
        """Run the full daily ingestion pipeline"""
        logger.info("Starting daily pipeline")
        
        # 1. Fetch new papers from arXiv
        papers = self.arxiv.fetch_daily_papers()
        logger.info(f"Fetched {len(papers)} papers")
        
        # 2. Curate and filter
        curated = self.curation.filter_and_rank(papers)
        logger.info(f"Curated {len(curated)} papers")
        
        # 3. Process each curated paper
        for paper in curated:
            self.process_paper(paper)
        
        logger.info("Daily pipeline complete")
    
    def process_paper(self, paper):
        """Process a single paper through the full pipeline"""
        logger.info(f"Processing: {paper.title}")
        
        # Download PDF
        pdf_path = self.arxiv.download_pdf(paper.id, f"/data/papers/{paper.id}/raw.pdf")
        
        # Extract text
        structured_text = self.pdf.extract_text_and_structure(pdf_path)
        
        # Chunk and embed
        chunks = self.pdf.chunk_for_embedding(structured_text)
        self.vector.embed_and_store(chunks)
        
        # Generate wiki
        wiki = self.wiki_extractor.generate(structured_text['full_text'])
        self._store_wiki(paper.id, wiki)
        
        # Extract claims
        claims = self.claim_extractor.extract(structured_text['full_text'], wiki)
        self._store_claims(paper.id, claims)
        
        # Create base artifact
        artifact = self.artifact_generator.create_base_artifact(
            paper.id, wiki, claims
        )
        
        logger.info(f"Completed processing: {paper.title}")
        return artifact
    
    def gap_analysis_pipeline(self):
        """Run gap detection across all papers"""
        logger.info("Starting gap analysis pipeline")
        
        all_gaps = []
        
        # Run each detector
        for detector in self.gap_detectors:
            gaps = detector.detect()
            all_gaps.extend(gaps)
            logger.info(f"{detector.__class__.__name__}: found {len(gaps)} gaps")
        
        # Deduplicate and rank
        all_gaps = self._deduplicate_gaps(all_gaps)
        all_gaps.sort(key=lambda g: g.get('total_score', 0), reverse=True)
        
        # Store gaps
        self._store_gaps(all_gaps)
        
        # Generate studies for top gaps
        top_gaps = all_gaps[:10]  # Top 10 gaps
        studies = []
        for gap in top_gaps:
            study = self.study_generator.create_study_artifact(gap, gap.get('scores', {}))
            studies.append(study)
        
        logger.info(f"Generated {len(studies)} studies from top gaps")
        
        return {
            'total_gaps_found': len(all_gaps),
            'studies_generated': len(studies),
            'top_gaps': top_gaps[:5],
            'top_studies': [s['artifact_id'] for s in studies]
        }
    
    def study_to_paper_pipeline(self, study_artifact_id):
        """Convert a completed study into a paper draft"""
        logger.info(f"Generating paper for study: {study_artifact_id}")
        
        # Get study with all experiment results
        study = self.db.get_artifact(study_artifact_id)
        
        # Verify all experiments complete
        incomplete = [e for e in study['experiments'] if e['status'] != 'completed']
        if incomplete:
            raise ValueError(f"Study has {len(incomplete)} incomplete experiments")
        
        # Generate paper draft
        paper_draft = self.paper_generator.generate_draft(study)
        
        # Update study artifact with paper draft
        study['paper_draft'] = paper_draft
        study['status'] = 'paper_drafted'
        study['version'] += 1
        
        self.db.update_artifact(study)
        
        # Generate LaTeX
        latex = self.paper_refiner.generate_latex(study)
        self._store_latex(study_artifact_id, latex)
        
        logger.info(f"Paper draft generated for study: {study_artifact_id}")
        
        return study
```

# electron-app/renderer/PaperStudio.jsx

function PaperStudio() {
    const [activeStudy, setActiveStudy] = useState(null);
    const [paperDraft, setPaperDraft] = useState(null);
    const [editingSection, setEditingSection] = useState(null);
    
    const handleRunMVP = async (experiment) => {
        setExperimentStatus('running');
        const result = await api.executeExperiment(
            activeStudy.artifact_id, 
            experiment.experiment_id,
            experiment.code
        );
        
        // Update local state
        setExperimentResult(result);
        
        // If MVP, show go/no-go analysis
        if (experiment.is_mvp) {
            showMVPAlert(result.go_no_go_analysis);
        }
    };
    
    const handleGeneratePaper = async () => {
        setPaperStatus('generating');
        const updatedStudy = await api.generatePaper(activeStudy.artifact_id);
        setPaperDraft(updatedStudy.paper_draft);
        setPaperStatus('draft_ready');
    };
    
    const handleRefineSection = async (sectionName, feedback) => {
        const updatedStudy = await api.refinePaperSection(
            activeStudy.artifact_id,
            sectionName,
            feedback
        );
        setPaperDraft(updatedStudy.paper_draft);
    };
    
    const handleExportLatex = async () => {
        const latex = await api.getLatex(activeStudy.artifact_id);
        // Save to file
        window.electronAPI.saveFile('paper.tex', latex);
    };
    
    return (
        <div className="paper-studio">
            <PaperStudioHeader study={activeStudy} />
            
            <div className="studio-layout">
                <PaperStudioSidebar 
                    paperDraft={paperDraft}
                    onSelectSection={setEditingSection}
                    studyProgress={activeStudy?.timeline}
                />
                
                <PaperStudioMain>
                    {editingSection ? (
                        <SectionEditor
                            section={editingSection}
                            content={paperDraft?.[editingSection]}
                            onSave={(content) => handleRefineSection(editingSection, content)}
                            onRequestAIRefinement={(feedback) => handleRefineSection(editingSection, feedback)}
                        />
                    ) : (
                        <PaperPreview paperDraft={paperDraft} />
                    )}
                </PaperStudioMain>
                
                <PaperStudioActions>
                    <ExperimentPanel
                        experiments={activeStudy?.experiments}
                        onRunExperiment={handleRunMVP}
                    />
                    <ActionPanel
                        onGeneratePaper={handleGeneratePaper}
                        onExportLatex={handleExportLatex}
                        paperStatus={paperStatus}
                    />
                </PaperStudioActions>
            </div>
        </div>
    );
}
