# backend/agents/verifiers/wiki_verifier.py

from typing import List, Dict, Optional, Tuple
from dataclasses import dataclass
from enum import Enum
import json
import logging

logger = logging.getLogger(__name__)

class VerificationStatus(Enum):
    SUPPORTED = "supported"
    UNSUPPORTED = "unsupported"
    PARTIALLY_SUPPORTED = "partially_supported"
    UNCLEAR = "unclear"
    ERROR = "error"

@dataclass
class ClaimVerification:
    claim: str
    status: VerificationStatus
    reasoning: str
    confidence: float  # 0.0 to 1.0
    source_evidence: Optional[str] = None
    llm_used: bool = True

class WikiVerifier:
    """
    Verifies that wiki claims are factually supported by the source paper text.
    Uses LLM-based verification with keyword overlap fallback.
    """
    
    def __init__(self, llm_provider, fallback_threshold: float = 0.4):
        self.llm = llm_provider
        self.fallback_threshold = fallback_threshold
        
    def verify_wiki_accuracy(
        self, 
        wiki_entry: Dict, 
        source_text: str,
        use_llm: bool = True
    ) -> Tuple[Dict, List[ClaimVerification]]:
        """
        Verify all factual claims in a wiki entry against source text.
        
        Returns:
            Updated wiki_entry with verification metadata
            List of individual claim verifications
        """
        # Extract all factual claims from the wiki
        claims = self._extract_claims(wiki_entry)
        
        # Verify each claim
        verifications = []
        unsupported_count = 0
        
        for claim in claims:
            if use_llm:
                try:
                    verification = self._verify_claim_with_llm(claim, source_text)
                except Exception as e:
                    logger.warning(f"LLM verification failed for claim: {e}")
                    verification = self._verify_claim_keyword_fallback(claim, source_text)
            else:
                verification = self._verify_claim_keyword_fallback(claim, source_text)
            
            verifications.append(verification)
            
            if verification.status in [VerificationStatus.UNSUPPORTED, VerificationStatus.ERROR]:
                unsupported_count += 1
        
        # Update wiki entry with verification metadata
        total_claims = len(claims) if claims else 1
        unsupported_ratio = unsupported_count / total_claims
        
        wiki_entry = dict(wiki_entry)  # Don't mutate input
        wiki_entry['_verification'] = {
            'total_claims': total_claims,
            'unsupported_claims': unsupported_count,
            'unsupported_ratio': unsupported_ratio,
            'needs_human_review': unsupported_ratio > 0.2,
            'verification_timestamp': self._now_iso(),
            'verifier_version': '2.0-llm-grounded'
        }
        
        return wiki_entry, verifications
    
    def _extract_claims(self, wiki_entry: Dict) -> List[str]:
        """Extract individual factual claims from wiki entry"""
        claims = []
        
        # Method claims
        if 'method' in wiki_entry:
            method = wiki_entry['method']
            if 'name' in method:
                claims.append(f"Method name: {method['name']}")
            if 'type' in method:
                claims.append(f"Method type: {method['type']}")
            claims.append(f"Method description: {method.get('high_level_description', '')}")
        
        # Key insights
        for insight in wiki_entry.get('key_insights', []):
            if insight.get('insight'):
                claims.append(insight['insight'])
        
        # Experiment results
        for result in wiki_entry.get('experiments', {}).get('key_results', []):
            if result.get('claim'):
                claims.append(result['claim'])
        
        # Limitations
        for limitation in wiki_entry.get('limitations', []):
            if limitation.get('limitation'):
                claims.append(limitation['limitation'])
        
        # Future work
        for fw in wiki_entry.get('future_work', []):
            if fw.get('direction'):
                claims.append(fw['direction'])
        
        return [c for c in claims if c.strip()]
    
    def _verify_claim_with_llm(
        self, 
        claim: str, 
        source_text: str
    ) -> ClaimVerification:
        """Use LLM to verify if a claim is supported by source text"""
        
        # Truncate source text if very long (use relevant chunks)
        source_excerpt = source_text[:15000] if len(source_text) > 15000 else source_text
        
        prompt = f"""You are a meticulous research fact-checker. 
Your task: determine if a specific claim is supported by the provided source text.

SOURCE TEXT (from academic paper):
{source_excerpt}

CLAIM TO VERIFY:
"{claim}"

Analyze carefully and provide your assessment:

1. Does the source text directly support this claim?
   - supported: The claim is explicitly stated or directly implied by the text
   - unsupported: The claim contradicts the text or is not mentioned
   - partially_supported: Some parts are supported, others are not
   - unclear: The text is ambiguous or insufficient to determine

2. Provide specific reasoning. Quote relevant text passages if applicable.

3. Rate your confidence (0.0 to 1.0) in your assessment.

Respond as JSON:
{{
    "status": "supported|unsupported|partially_supported|unclear",
    "reasoning": "Detailed explanation",
    "source_evidence": "Relevant quote from source text, or null",
    "confidence": 0.85
}}
"""
        
        response = self.llm.complete(prompt, response_format="json")
        result = json.loads(response)
        
        return ClaimVerification(
            claim=claim,
            status=VerificationStatus(result['status']),
            reasoning=result['reasoning'],
            confidence=result['confidence'],
            source_evidence=result.get('source_evidence'),
            llm_used=True
        )
    
    def _verify_claim_keyword_fallback(
        self, 
        claim: str, 
        source_text: str
    ) -> ClaimVerification:
        """Keyword-based fallback when LLM is unavailable"""
        claim_words = set(claim.lower().split())
        source_words = set(source_text.lower().split())
        
        if not claim_words:
            return ClaimVerification(
                claim=claim,
                status=VerificationStatus.UNCLEAR,
                reasoning="Empty claim",
                confidence=0.0,
                llm_used=False
            )
        
        overlap = len(claim_words & source_words) / len(claim_words)
        
        if overlap >= self.fallback_threshold:
            status = VerificationStatus.SUPPORTED
            confidence = min(overlap, 0.7)  # Cap keyword confidence at 0.7
        else:
            status = VerificationStatus.UNSUPPORTED
            confidence = max(1.0 - overlap, 0.1)
        
        return ClaimVerification(
            claim=claim,
            status=status,
            reasoning=f"Keyword overlap: {overlap:.2%}",
            confidence=confidence,
            llm_used=False
        )
    
    def _now_iso(self) -> str:
        from datetime import datetime, timezone
        return datetime.now(timezone.utc).isoformat()

# backend/agents/detectors/contradiction_detector.py (deepened)

from typing import List, Dict, Optional, Tuple
from dataclasses import dataclass
from enum import Enum
import json
import logging

logger = logging.getLogger(__name__)

class ContradictionType(Enum):
    GENUINE = "genuine_contradiction"
    DIFFERENT_CONDITIONS = "different_conditions"
    INCOMPARABLE = "incomparable"
    SAME_RESULT = "same_result"
    UNCERTAIN = "uncertain"

@dataclass
class ContradictionVerification:
    claim_a: Dict
    claim_b: Dict
    contradiction_type: ContradictionType
    reasoning: str
    confidence: float
    possible_explanations: List[str]
    resolution_suggested: Optional[str] = None
    llm_used: bool = True

class ContradictionDetector:
    """
    Detects genuine contradictions between research claims.
    Uses LLM to distinguish real contradictions from different experimental conditions.
    """
    
    def __init__(self, llm_provider, db_session, numeric_threshold: float = 0.1):
        self.llm = llm_provider
        self.db = db_session
        self.numeric_threshold = numeric_threshold
    
    def detect(self) -> List[Dict]:
        """Find and verify potential contradictions across papers"""
        # Stage 1: Find candidate pairs (same metric + dataset + method)
        candidates = self._find_candidate_contradictions()
        
        # Stage 2: LLM verification of each candidate
        verified = []
        for candidate in candidates:
            verification = self._verify_contradiction(
                candidate['claim_a'],
                candidate['claim_b']
            )
            
            if verification.contradiction_type == ContradictionType.GENUINE:
                # Check if any paper has reconciled these
                reconciliation = self._find_reconciliation(
                    candidate['paper_a_id'],
                    candidate['paper_b_id']
                )
                
                if not reconciliation:
                    gap = self._create_gap_from_contradiction(
                        candidate, verification
                    )
                    verified.append(gap)
        
        return verified
    
    def _verify_contradiction(
        self, 
        claim_a: Dict, 
        claim_b: Dict,
        use_llm: bool = True
    ) -> ContradictionVerification:
        """
        Verify if two claims genuinely contradict.
        Uses LLM for context-aware judgment, falls back to numeric heuristic.
        """
        
        if use_llm:
            try:
                return self._verify_with_llm(claim_a, claim_b)
            except Exception as e:
                logger.warning(f"LLM contradiction verification failed: {e}")
                return self._verify_with_heuristic(claim_a, claim_b)
        else:
            return self._verify_with_heuristic(claim_a, claim_b)
    
    def _verify_with_llm(
        self, 
        claim_a: Dict, 
        claim_b: Dict
    ) -> ContradictionVerification:
        """LLM-based contradiction verification"""
        
        prompt = f"""You are a research integrity analyst. Determine if two research findings genuinely contradict each other.

CLAIM A:
- Paper: {claim_a.get('paper_title', 'Unknown')}
- Method: {claim_a.get('method_name', 'Unknown')}
- Dataset: {claim_a.get('dataset', 'Unknown')}
- Metric: {claim_a.get('metric', 'Unknown')}
- Value: {claim_a.get('value', 'Unknown')}
- Task/Direction: {claim_a.get('task', 'Not specified')}
- Language pair (if translation): {claim_a.get('language_pair', 'Not specified')}
- Experimental details: {claim_a.get('details', 'Not specified')}

CLAIM B:
- Paper: {claim_b.get('paper_title', 'Unknown')}
- Method: {claim_b.get('method_name', 'Unknown')}
- Dataset: {claim_b.get('dataset', 'Unknown')}
- Metric: {claim_b.get('metric', 'Unknown')}
- Value: {claim_b.get('value', 'Unknown')}
- Task/Direction: {claim_b.get('task', 'Not specified')}
- Language pair (if translation): {claim_b.get('language_pair', 'Not specified')}
- Experimental details: {claim_b.get('details', 'Not specified')}

Classify the relationship between these claims:

1. GENUINE CONTRADICTION: Same method, same task, same experimental conditions produce meaningfully different results
2. DIFFERENT CONDITIONS: Different experimental setups explain the difference (different dataset splits, different hyperparameters, different languages, different hardware)
3. INCOMPARABLE: Claims measure fundamentally different things or use incomparable metrics
4. SAME RESULT: Differences are within statistical noise or measurement error
5. UNCERTAIN: Not enough information to determine

Key considerations:
- Same BLEU score for EN→DE vs FR→EN is NOT a contradiction (different language pairs)
- Same accuracy on different test sets is NOT necessarily a contradiction
- Different hyperparameters or implementation details can explain small differences
- Only flag as GENUINE CONTRADICTION if the difference is unlikely to be explained by experimental variation

Respond as JSON:
{{
    "contradiction_type": "genuine_contradiction|different_conditions|incomparable|same_result|uncertain",
    "reasoning": "Detailed explanation of your classification",
    "confidence": 0.0-1.0,
    "possible_explanations": ["List of alternative explanations for the difference"],
    "resolution_suggested": "How to determine the truth, or null",
    "key_differences": ["List of critical differences between the two experimental setups"]
}}
"""
        
        response = self.llm.complete(prompt, response_format="json")
        result = json.loads(response)
        
        return ContradictionVerification(
            claim_a=claim_a,
            claim_b=claim_b,
            contradiction_type=ContradictionType(result['contradiction_type']),
            reasoning=result['reasoning'],
            confidence=result['confidence'],
            possible_explanations=result['possible_explanations'],
            resolution_suggested=result.get('resolution_suggested'),
            llm_used=True
        )
    
    def _verify_with_heuristic(
        self, 
        claim_a: Dict, 
        claim_b: Dict
    ) -> ContradictionVerification:
        """Numeric heuristic fallback"""
        val_a = self._extract_numeric(claim_a.get('value'))
        val_b = self._extract_numeric(claim_b.get('value'))
        
        explanations = [
            "Different experimental conditions may explain the difference",
            "Different implementation details could cause variation"
        ]
        
        if val_a is None or val_b is None:
            return ContradictionVerification(
                claim_a=claim_a,
                claim_b=claim_b,
                contradiction_type=ContradictionType.UNCERTAIN,
                reasoning="Cannot extract numeric values for comparison",
                confidence=0.2,
                possible_explanations=explanations,
                llm_used=False
            )
        
        diff = abs(val_a - val_b) / max(abs(val_a), abs(val_b)) if max(abs(val_a), abs(val_b)) > 0 else 0
        
        if diff > self.numeric_threshold:
            return ContradictionVerification(
                claim_a=claim_a,
                claim_b=claim_b,
                contradiction_type=ContradictionType.GENUINE,
                reasoning=f"Numeric difference of {diff:.1%} exceeds threshold",
                confidence=min(diff / 0.2, 0.7),
                possible_explanations=explanations,
                llm_used=False
            )
        else:
            return ContradictionVerification(
                claim_a=claim_a,
                claim_b=claim_b,
                contradiction_type=ContradictionType.SAME_RESULT,
                reasoning=f"Numeric difference of {diff:.1%} within threshold",
                confidence=0.5,
                possible_explanations=[],
                llm_used=False
            )
    
    def _extract_numeric(self, value) -> Optional[float]:
        """Extract numeric value from various formats"""
        if value is None:
            return None
        if isinstance(value, (int, float)):
            return float(value)
        import re
        match = re.search(r'[\d.]+', str(value))
        return float(match.group()) if match else None
    
    def _find_candidate_contradictions(self) -> List[Dict]:
        """Find claim pairs that might contradict (same metric + dataset)"""
        return self.db.query("""
            SELECT 
                c1.content as claim_a, c2.content as claim_b,
                c1.paper_id as paper_a_id, c2.paper_id as paper_b_id,
                p1.title as paper_a_title, p2.title as paper_b_title
            FROM claims c1
            JOIN claims c2 ON 
                c1.claim_type = 'result' 
                AND c2.claim_type = 'result'
                AND c1.content->>'metric' = c2.content->>'metric'
                AND c1.content->>'dataset' = c2.content->>'dataset'
                AND c1.content->>'method_name' = c2.content->>'method_name'
                AND c1.paper_id < c2.paper_id
            JOIN papers p1 ON c1.paper_id = p1.id
            JOIN papers p2 ON c2.paper_id = p2.id
        """).fetchall()
    
    def _find_reconciliation(self, paper_id_a: str, paper_id_b: str) -> bool:
        """Check if any paper cites both and reconciles them"""
        # Implementation depends on citation graph in your DB
        return False  # Simplified
    
    def _create_gap_from_contradiction(self, candidate, verification) -> Dict:
        """Create a gap entry from a verified contradiction"""
        gap = {
            'gap_type': 'contradiction',
            'title': f"Resolve contradiction: {candidate['claim_a'].get('method_name')} on {candidate['claim_a'].get('dataset')}",
            'claim_a': candidate['claim_a'],
            'claim_b': candidate['claim_b'],
            'verification': {
                'type': verification.contradiction_type.value,
                'reasoning': verification.reasoning,
                'confidence': verification.confidence,
                'possible_explanations': verification.possible_explanations,
                'resolution_suggested': verification.resolution_suggested
            },
            'papers_involved': [
                candidate['paper_a_id'],
                candidate['paper_b_id']
            ]
        }
        return gap

# backend/agents/detectors/method_problem_detector.py (deepened)

from typing import List, Dict, Optional, Tuple
from dataclasses import dataclass
import json
import logging

logger = logging.getLogger(__name__)

@dataclass
class ApplicabilityAssessment:
    method_name: str
    problem_dataset: str
    applicability_score: float  # 0.0 to 1.0
    requires_adaptation: bool
    adaptations_needed: List[str]
    estimated_improvement: Optional[str]  # e.g., "10-20% speed improvement expected"
    rationale: str
    risk_factors: List[str]
    confidence: float
    llm_used: bool = True

class MethodProblemGapDetector:
    """
    Identifies gaps where methods haven't been applied to problems.
    Uses LLM to score applicability instead of hardcoded values.
    """
    
    def __init__(self, llm_provider, db_session):
        self.llm = llm_provider
        self.db = db_session
    
    def detect(self) -> List[Dict]:
        """Find method-problem gaps with LLM-scored applicability"""
        methods = self._get_all_methods()
        problems = self._get_all_problems()
        existing_pairs = self._get_existing_pairs()
        
        gaps = []
        for method in methods:
            for problem in problems:
                pair_key = (method['name'], problem['dataset'])
                
                if pair_key not in existing_pairs:
                    # Score using LLM
                    assessment = self._assess_applicability(method, problem)
                    
                    if assessment.applicability_score > 0.3:  # Minimum threshold
                        gap = self._build_gap(method, problem, assessment)
                        gaps.append(gap)
        
        return sorted(gaps, key=lambda g: g['applicability_score'], reverse=True)
    
    def _assess_applicability(
        self, 
        method: Dict, 
        problem: Dict,
        use_llm: bool = True
    ) -> ApplicabilityAssessment:
        """
        Assess if a method could be applied to a problem.
        Uses LLM for nuanced assessment, falls back to heuristic.
        """
        
        if use_llm:
            try:
                return self._assess_with_llm(method, problem)
            except Exception as e:
                logger.warning(f"LLM applicability assessment failed: {e}")
                return self._assess_with_heuristic(method, problem)
        else:
            return self._assess_with_heuristic(method, problem)
    
    def _assess_with_llm(self, method: Dict, problem: Dict) -> ApplicabilityAssessment:
        """LLM-based applicability assessment"""
        
        prompt = f"""You are a machine learning research advisor. Assess whether a specific method 
could be effectively applied to a specific problem/dataset.

METHOD:
- Name: {method.get('name', 'Unknown')}
- Category: {method.get('category', 'Unknown')}
- Description: {method.get('description', 'No description available')}
- Modality: {method.get('modality', 'Not specified')}
- Assumptions: {method.get('assumptions', ['None specified'])}
- Computational requirements: {method.get('constraints', {})}
- Source paper: {method.get('source_paper', 'Unknown')}

PROBLEM / DATASET:
- Name: {problem.get('dataset', 'Unknown')}
- Task type: {problem.get('task_type', 'Not specified')}
- Data modality: {problem.get('modality', 'Not specified')}
- Scale: {problem.get('sample_size', 'Unknown')} samples
- Domain: {problem.get('domain', 'General')}
- Current best method: {problem.get('sota_method', 'Unknown')}
- Current best performance: {problem.get('sota_performance', 'Unknown')}

Assess the following:

1. APPLICABILITY SCORE (0.0-1.0):
   - 0.0-0.2: Fundamentally incompatible (wrong modality, different problem class)
   - 0.3-0.5: Possibly applicable but requires major adaptation
   - 0.6-0.8: Likely applicable with minor modifications
   - 0.9-1.0: Directly applicable, clear fit

2. ADAPTATIONS NEEDED: What changes would be required?

3. ESTIMATED IMPROVEMENT: If successful, what kind of improvement might be expected over current approaches? (Be specific: "10-20% accuracy improvement" or "2-3x speedup")

4. RISK FACTORS: What could prevent this from working?

5. RATIONALE: Why do you think this would or wouldn't work?

Key considerations:
- BERT (text) on ImageNet (images) would score LOW (modality mismatch without adaptation)
- ViT (vision transformer) on text classification would score MEDIUM (architecture adaptable)
- FlashAttention (efficient attention) on long-document tasks would score HIGH (clear benefit)

Respond as JSON:
{{
    "applicability_score": 0.65,
    "requires_adaptation": true,
    "adaptations_needed": ["List specific changes required"],
    "estimated_improvement": "Specific, quantitative estimate if possible",
    "rationale": "Detailed reasoning",
    "risk_factors": ["Potential failure modes"],
    "confidence": 0.85
}}
"""
        
        response = self.llm.complete(prompt, response_format="json")
        result = json.loads(response)
        
        return ApplicabilityAssessment(
            method_name=method['name'],
            problem_dataset=problem['dataset'],
            applicability_score=result['applicability_score'],
            requires_adaptation=result.get('requires_adaptation', False),
            adaptations_needed=result.get('adaptations_needed', []),
            estimated_improvement=result.get('estimated_improvement'),
            rationale=result['rationale'],
            risk_factors=result.get('risk_factors', []),
            confidence=result.get('confidence', 0.5),
            llm_used=True
        )
    
    def _assess_with_heuristic(
        self, 
        method: Dict, 
        problem: Dict
    ) -> ApplicabilityAssessment:
        """Heuristic-based assessment (fallback)"""
        
        # Basic modality matching
        method_modality = method.get('modality', '').lower()
        problem_modality = problem.get('modality', '').lower()
        
        # Category compatibility
        method_category = method.get('category', '').lower()
        problem_task = problem.get('task_type', '').lower()
        
        score = 0.5  # Default, not very informative
        
        # Modality match bonus
        if method_modality and problem_modality:
            if method_modality == problem_modality:
                score = 0.7
            elif any(m in problem_modality for m in method_modality.split(',')):
                score = 0.6
            else:
                score = 0.3
        
        # Task match bonus
        task_compatible_categories = {
            'classification': ['classification', 'supervised_learning'],
            'generation': ['generation', 'language_modeling'],
            'translation': ['seq2seq', 'transformer'],
            'detection': ['object_detection', 'vision'],
        }
        
        for task_key, compatible_cats in task_compatible_categories.items():
            if task_key in problem_task and any(c in method_category for c in compatible_cats):
                score = min(score + 0.2, 1.0)
        
        return ApplicabilityAssessment(
            method_name=method['name'],
            problem_dataset=problem['dataset'],
            applicability_score=score,
            requires_adaptation=score < 0.7,
            adaptations_needed=['Check modality compatibility'] if score < 0.7 else [],
            estimated_improvement=None,
            rationale=f"Heuristic assessment: modality match={method_modality == problem_modality}, score={score:.2f}",
            risk_factors=['Heuristic assessment may be inaccurate'],
            confidence=0.3,
            llm_used=False
        )
    
    def _build_gap(self, method, problem, assessment) -> Dict:
        """Build a gap entry from assessment"""
        return {
            'gap_type': 'method_problem',
            'method': method,
            'problem': problem,
            'applicability_score': assessment.applicability_score,
            'assessment': {
                'rationale': assessment.rationale,
                'adaptations_needed': assessment.adaptations_needed,
                'estimated_improvement': assessment.estimated_improvement,
                'risk_factors': assessment.risk_factors,
                'confidence': assessment.confidence
            },
            'title': f"Apply {method['name']} to {problem['dataset']}"
        }
    
    def _get_all_methods(self) -> List[Dict]:
        return self.db.query("""
            SELECT DISTINCT 
                content->>'name' as name,
                content->>'category' as category,
                content->>'description' as description,
                content->>'modality' as modality,
                content->>'assumptions' as assumptions,
                content->'constraints' as constraints,
                paper_id as source_paper
            FROM claims 
            WHERE claim_type = 'method'
        """).fetchall()
    
    def _get_all_problems(self) -> List[Dict]:
        return self.db.query("""
            SELECT DISTINCT
                content->>'dataset' as dataset,
                content->>'task_type' as task_type,
                content->>'modality' as modality,
                content->>'sample_size' as sample_size,
                content->>'domain' as domain,
                content->>'sota_method' as sota_method,
                content->>'sota_performance' as sota_performance
            FROM claims
            WHERE claim_type = 'result'
        """).fetchall()
    
    def _get_existing_pairs(self) -> set:
        rows = self.db.query("""
            SELECT 
                content->>'method_name' as method,
                content->>'dataset' as dataset
            FROM claims
            WHERE claim_type = 'result'
        """).fetchall()
        return {(r['method'], r['dataset']) for r in rows}

# backend/agents/generators/connection_agent.py (deepened)

from typing import List, Dict, Optional, Set, Tuple
from dataclasses import dataclass
from enum import Enum
import json
import logging

logger = logging.getLogger(__name__)

class ConnectionType(Enum):
    COMPLEMENTS = "complements"
    CONTRADICTS = "contradicts"
    EXTENDS = "extends"
    ALTERNATIVE_APPROACH = "alternative_approach"
    SHARES_INSIGHT = "shares_insight"
    ENABLES = "enables"
    REPRODUCES = "reproduces"
    SYNTHESIS_OPPORTUNITY = "synthesis_opportunity"

@dataclass
class PaperConnection:
    source_paper_id: str
    target_paper_id: str
    connection_type: ConnectionType
    confidence: float
    evidence: str
    explanation: str
    potential_synthesis: Optional[str] = None
    llm_inferred: bool = True

class ConnectionAgent:
    """
    Finds meaningful connections between papers beyond explicit citations.
    Uses LLM to infer implicit relationships from method and result overlaps.
    """
    
    def __init__(self, llm_provider, db_session, vector_service):
        self.llm = llm_provider
        self.db = db_session
        self.vector = vector_service
    
    def find_connections(
        self, 
        paper_id: str, 
        candidate_papers: List[str] = None,
        use_llm: bool = True
    ) -> List[PaperConnection]:
        """
        Find all meaningful connections between a paper and others in the library.
        
        Args:
            paper_id: The paper to find connections for
            candidate_papers: Optional list of candidate paper IDs to check
            use_llm: Whether to use LLM for inference
        """
        
        if candidate_papers is None:
            candidate_papers = self._get_candidate_papers(paper_id)
        
        connections = []
        for target_id in candidate_papers:
            connection = self._analyze_pair(
                paper_id, 
                target_id, 
                use_llm=use_llm
            )
            if connection and connection.confidence > 0.5:
                connections.append(connection)
        
        return sorted(connections, key=lambda c: c.confidence, reverse=True)
    
    def _analyze_pair(
        self, 
        source_id: str, 
        target_id: str,
        use_llm: bool = True
    ) -> Optional[PaperConnection]:
        """Analyze the relationship between two papers"""
        
        # Get structured data for both papers
        source = self._get_paper_data(source_id)
        target = self._get_paper_data(target_id)
        
        if not source or not target:
            return None
        
        # Check for explicit connections first
        explicit = self._check_explicit_connections(source, target)
        if explicit:
            return explicit
        
        # If no explicit connection, use LLM for implicit inference
        if use_llm:
            try:
                return self._infer_connection_with_llm(source, target)
            except Exception as e:
                logger.warning(f"LLM connection inference failed: {e}")
                return self._infer_connection_heuristic(source, target)
        else:
            return self._infer_connection_heuristic(source, target)
    
    def _infer_connection_with_llm(
        self, 
        source: Dict, 
        target: Dict
    ) -> Optional[PaperConnection]:
        """Use LLM to infer implicit connections between papers"""
        
        prompt = f"""You are a research synthesis analyst. Identify meaningful connections 
between two research papers that may not be explicitly cited but have important 
relationships.

PAPER A:
Title: {source.get('title', 'Unknown')}
Methods: {json.dumps(source.get('methods', []), indent=2)}
Key results: {json.dumps(source.get('results', []), indent=2)}
Claims: {json.dumps(source.get('claims', [])[:5], indent=2)}
Limitations: {json.dumps(source.get('limitations', []), indent=2)}

PAPER B:
Title: {target.get('title', 'Unknown')}
Methods: {json.dumps(target.get('methods', []), indent=2)}
Key results: {json.dumps(target.get('results', []), indent=2)}
Claims: {json.dumps(target.get('claims', [])[:5], indent=2)}
Limitations: {json.dumps(target.get('limitations', []), indent=2)}

Analyze the relationship between these papers. Consider:

1. COMPLEMENTS: Do they address complementary aspects of the same problem?
2. ENABLES: Does one paper's method enable or unlock the other's approach?
3. SYNTHESIS_OPPORTUNITY: Could combining insights from both produce something new?
4. ALTERNATIVE_APPROACH: Do they solve the same problem differently?
5. SHARES_INSIGHT: Do they converge on the same insight from different angles?
6. EXTENDS: Could one's method naturally extend the other's?
7. CONTRADICTS: Do their findings challenge each other?
8. REPRODUCES: Does one reproduce or validate the other's claims?

Focus on NON-OBVIOUS connections that someone reading both papers might miss.
Look beyond keyword overlap to deeper conceptual relationships.

Respond as JSON:
{{
    "has_meaningful_connection": true/false,
    "connection_type": "complements|enables|synthesis_opportunity|alternative_approach|shares_insight|extends|contradicts|reproduces",
    "confidence": 0.0-1.0,
    "explanation": "Detailed explanation of the relationship",
    "evidence": "Specific overlapping methods, results, or insights that support this connection",
    "potential_synthesis": "If these were combined, what new capability would emerge?",
    "key_overlaps": ["Specific shared elements"],
    "key_differences": ["Critical differences that make synthesis interesting"]
}}

If no meaningful connection exists, return has_meaningful_connection: false.
"""
        
        response = self.llm.complete(prompt, response_format="json")
        result = json.loads(response)
        
        if not result.get('has_meaningful_connection', False):
            return None
        
        return PaperConnection(
            source_paper_id=source['id'],
            target_paper_id=target['id'],
            connection_type=ConnectionType(result['connection_type']),
            confidence=result['confidence'],
            evidence=result['evidence'],
            explanation=result['explanation'],
            potential_synthesis=result.get('potential_synthesis'),
            llm_inferred=True
        )
    
    def _check_explicit_connections(
        self, 
        source: Dict, 
        target: Dict
    ) -> Optional[PaperConnection]:
        """Check for explicit connections from extracted claims"""
        
        source_claims = source.get('claims', [])
        
        for claim in source_claims:
            if claim.get('type') == 'comparison':
                if claim.get('compared_to') and target['id'] in str(claim['compared_to']):
                    return PaperConnection(
                        source_paper_id=source['id'],
                        target_paper_id=target['id'],
                        connection_type=self._map_relationship(claim.get('relationship', '')),
                        confidence=0.9,
                        evidence=f"Explicit comparison: {claim.get('specific_difference', '')}",
                        explanation=claim.get('specific_difference', 'Explicitly compared in paper'),
                        llm_inferred=False
                    )
        
        return None
    
    def _infer_connection_heuristic(
        self, 
        source: Dict, 
        target: Dict
    ) -> Optional[PaperConnection]:
        """Heuristic fallback based on method and problem overlap"""
        
        source_methods = {m.get('name', '').lower() for m in source.get('methods', [])}
        target_methods = {m.get('name', '').lower() for m in target.get('methods', [])}
        
        source_datasets = {r.get('dataset', '').lower() for r in source.get('results', [])}
        target_datasets = {r.get('dataset', '').lower() for r in target.get('results', [])}
        
        shared_methods = source_methods & target_methods
        shared_datasets = source_datasets & target_datasets
        
        if shared_methods and shared_datasets:
            return PaperConnection(
                source_paper_id=source['id'],
                target_paper_id=target['id'],
                connection_type=ConnectionType.SHARES_INSIGHT,
                confidence=0.4,
                evidence=f"Both use {shared_methods} on {shared_datasets}",
                explanation="Heuristic: shared methods and datasets suggest related work",
                llm_inferred=False
            )
        
        if shared_methods:
            return PaperConnection(
                source_paper_id=source['id'],
                target_paper_id=target['id'],
                connection_type=ConnectionType.ALTERNATIVE_APPROACH,
                confidence=0.3,
                evidence=f"Both use {shared_methods}",
                explanation="Heuristic: shared methods",
                llm_inferred=False
            )
        
        return None
    
    def _get_paper_data(self, paper_id: str) -> Optional[Dict]:
        """Get structured paper data"""
        wiki = self.db.query(
            "SELECT * FROM wiki_entries WHERE paper_id = :pid",
            {'pid': paper_id}
        ).fetchone()
        
        claims = self.db.query(
            "SELECT * FROM claims WHERE paper_id = :pid",
            {'pid': paper_id}
        ).fetchall()
        
        if not wiki:
            return None
        
        return {
            'id': paper_id,
            'title': wiki.get('title', ''),
            'methods': [c['content'] for c in claims if c['claim_type'] == 'method'],
            'results': [c['content'] for c in claims if c['claim_type'] == 'result'],
            'claims': [c['content'] for c in claims],
            'limitations': [c['content'] for c in claims if c['claim_type'] == 'limitation']
        }
    
    def _get_candidate_papers(self, paper_id: str) -> List[str]:
        """Get candidate papers for connection analysis"""
        # Vector search for semantically similar papers
        similar = self.vector.search_similar(paper_id, top_k=20)
        return [p['paper_id'] for p in similar if p['paper_id'] != paper_id]
    
    def _map_relationship(self, relationship: str) -> ConnectionType:
        mapping = {
            'improves_on': ConnectionType.EXTENDS,
            'different_approach': ConnectionType.ALTERNATIVE_APPROACH,
            'contradicts': ConnectionType.CONTRADICTS,
            'complements': ConnectionType.COMPLEMENTS
        }
        return mapping.get(relationship, ConnectionType.SHARES_INSIGHT)

        metrics['method_problem_differentiation'] = \
            1.0 if assessment_high.applicability_score > assessment_low.applicability_score else 0.0
        
        # 4. Study grounding
        designer = StudyDesigner(llm_provider)
        gap = {
            'method': {'name': 'FlashAttention'},

        metrics['study_grounding_score'] = \
            1.0 if 'flashattention' in hypothesis_lower or 'flash' in hypothesis_lower else 0.0

        agent = ConnectionAgent(llm_provider, MockDB(), MockVector())

### Key Design Patterns Applied

1. **Try/Except with Graceful Degradation**: Every module attempts LLM first, falls back to heuristic on failure. The system never breaks when the LLM is unavailable.

2. **Quality Tests as Specifications**: Each batch includes three test types:
   - **Happy path**: Real LLM produces correct, high-quality output
   - **Fallback**: Works correctly when LLM fails
   - **Quality**: Intentional wrong input is caught and rejected

3. **Structured Output Contracts**: All LLM responses use JSON with defined schemas, making parsing reliable and type-safe.

4. **Confidence Scoring**: Every verification includes confidence scores, allowing downstream systems to weight or filter based on certainty.
